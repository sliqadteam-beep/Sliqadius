(()=>{

  const mMetalDark=new THREE.MeshStandardMaterial({color:0x626a6f,metalness:.55,roughness:.54});

  };

    if(c.k===K.HOSP){const red=new THREE.MeshStandardMaterial({color:0xd94a4f,emissive:0x7a1619,emissiveIntensity:.25});g.add(box(.38,.06,.06,red,wx,.88,wz+S*.37));g.add(box(.06,.38,.06,red,wx,.88,wz+S*.37));}
    if(c.k===K.LANDFILL){for(let i=0;i<5;i++){let bag=new THREE.Mesh(new THREE.SphereGeometry(.11+Math.random()*.07,7,5),new THREE.MeshStandardMaterial({color:[0x53584b,0x6c5e4e,0x77705c][i%3],roughness:1}));bag.scale.y=.7;bag.position.set(wx-.7+Math.random()*1.4,.09,wz-.6+Math.random()*1.2);g.add(bag)}}
