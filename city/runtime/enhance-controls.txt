const ctlKeys=new Set();
addEventListener('keydown',e=>{
  const k=e.key.toLowerCase(); ctlKeys.add(k);
  if(['w','a','s','d','q','e','arrowup','arrowdown','arrowleft','arrowright','+','=','-','escape','home'].includes(k)) e.preventDefault();
  if(k==='escape'){drag=null;orbit=false;}
  if(k==='home'){target.set(0,0,0);dist=78;az=.78;el=.78;}
});
addEventListener('keyup',e=>ctlKeys.delete(e.key.toLowerCase()));
let ctlLast=performance.now();
function controlLoop(t){
  requestAnimationFrame(controlLoop);
  let dt=Math.min(.05,(t-ctlLast)/1000||.016);ctlLast=t;
  let sp=(ctlKeys.has('shift')?42:22)*dt*(.55+dist/90);
  let fx=-Math.cos(az),fz=-Math.sin(az),rx=-fz,rz=fx;
  let mx=0,mz=0;
  if(ctlKeys.has('w')||ctlKeys.has('arrowup')){mx+=fx;mz+=fz}
  if(ctlKeys.has('s')||ctlKeys.has('arrowdown')){mx-=fx;mz-=fz}
  if(ctlKeys.has('a')||ctlKeys.has('arrowleft')){mx-=rx;mz-=rz}
  if(ctlKeys.has('d')||ctlKeys.has('arrowright')){mx+=rx;mz+=rz}
  if(mx||mz){let l=Math.hypot(mx,mz)||1;target.x=Math.max(-H,Math.min(H,target.x+mx/l*sp));target.z=Math.max(-H,Math.min(H,target.z+mz/l*sp));}
  if(ctlKeys.has('q'))az+=1.35*dt;
  if(ctlKeys.has('e'))az-=1.35*dt;
  if(ctlKeys.has('+')||ctlKeys.has('='))dist=Math.max(24,dist-35*dt);
  if(ctlKeys.has('-'))dist=Math.min(150,dist+35*dt);
}
requestAnimationFrame(controlLoop);
