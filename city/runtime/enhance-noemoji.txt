// Strip emojis from all current and future DOM text.
function __stripEmojiDOM(root=document.body){
 const w=document.createTreeWalker(root,NodeFilter.SHOW_TEXT);let n;
 while(n=w.nextNode()){
  const c=__cleanText(n.nodeValue);
  if(c!==n.nodeValue)n.nodeValue=c;
 }
}
__stripEmojiDOM();
const __emojiObserver=new MutationObserver(ms=>{
 for(const m of ms){
  for(const n of m.addedNodes){
   if(n.nodeType===3){const c=__cleanText(n.nodeValue);if(c!==n.nodeValue)n.nodeValue=c}
   else if(n.nodeType===1)__stripEmojiDOM(n);
  }
  if(m.type==='characterData'){const c=__cleanText(m.target.nodeValue);if(c!==m.target.nodeValue)m.target.nodeValue=c}
 }
});
__emojiObserver.observe(document.body,{subtree:true,childList:true,characterData:true});
